## Overview

Create comprehensive unit tests for Swift and Kotlin bindings generated by UniFFI. This validates that the FFI layer correctly exposes Rust functionality to mobile platforms.

**Current State**: UniFFI generates Swift and Kotlin bindings, but platform-specific compilation and unit tests are not implemented.

**Beta Exit Criterion**: "Swift/Kotlin bindings pass unit tests"

```swift
import XCTest
@testable import LexiconSwift

final class LexiconContextTests: XCTestCase {

    var tempDir: URL!

    override func setUp() {
        super.setUp()
        tempDir = FileManager.default.temporaryDirectory
            .appendingPathComponent(UUID().uuidString)
        try? FileManager.default.createDirectory(at: tempDir, withIntermediateDirectories: true)
    }

    override func tearDown() {
        try? FileManager.default.removeItem(at: tempDir)
        super.tearDown()
    }

    // MARK: - Context Creation

    func testContextCreation() throws {
        let config = LexiconConfig(
            modelDir: tempDir.appendingPathComponent("models"),
            dataDir: tempDir.appendingPathComponent("data")
        )

        let context = try LexiconContext(config: config)
        XCTAssertNotNil(context)
    }

    func testContextCreationWithInvalidPath() {
        let config = LexiconConfig(
            modelDir: URL(fileURLWithPath: "/nonexistent/path"),
            dataDir: URL(fileURLWithPath: "/nonexistent/path")
        )

        XCTAssertThrowsError(try LexiconContext(config: config)) { error in
            XCTAssertTrue(error is LexiconError)
        }
    }

    // MARK: - Error Mapping

    func testErrorMapping() {
        // Verify Rust errors map to Swift exceptions
        // Test each error variant
    }

    // MARK: - Memory Management

    func testNoMemoryLeaks() {
        weak var weakContext: LexiconContext?

        autoreleasepool {
            let config = LexiconConfig(modelDir: tempDir, dataDir: tempDir)
            let context = try? LexiconContext(config: config)
            weakContext = context
            XCTAssertNotNil(weakContext)
        }

        // Context should be deallocated
        XCTAssertNil(weakContext)
    }

    func testRepeatedOperations() throws {
        let config = LexiconConfig(modelDir: tempDir, dataDir: tempDir)

        for _ in 0..<100 {
            let context = try LexiconContext(config: config)
            _ = context.availableModels()
        }
        // Should not crash or leak
    }

    // MARK: - Async Operations

    func testAsyncModelLoading() async throws {
        let config = LexiconConfig(modelDir: tempDir, dataDir: tempDir)
        let context = try LexiconContext(config: config)

        // Note: This will fail without actual model, but tests async path
        do {
            try await context.loadModel(modelId: "nonexistent")
            XCTFail("Should throw for nonexistent model")
        } catch {
            XCTAssertTrue(error is LexiconError)
        }
    }

    // MARK: - Thread Safety

    func testConcurrentAccess() throws {
        let config = LexiconConfig(modelDir: tempDir, dataDir: tempDir)
        let context = try LexiconContext(config: config)

        let expectation = expectation(description: "Concurrent access")
        expectation.expectedFulfillmentCount = 10

        for _ in 0..<10 {
            DispatchQueue.global().async {
                _ = context.availableModels()
                expectation.fulfill()
            }
        }

        wait(for: [expectation], timeout: 5.0)
    }
}
```

### Kotlin Test Suite

```kotlin
// src/test/kotlin/com/lexicon/core/LexiconContextTest.kt

package com.lexicon.core

import org.junit.After
import org.junit.Before
import org.junit.Test
import org.junit.Assert.*
import java.io.File
import java.util.UUID
import kotlinx.coroutines.runBlocking
import kotlinx.coroutines.async
import kotlinx.coroutines.awaitAll

class LexiconContextTest {

    private lateinit var tempDir: File

    @Before
    fun setUp() {
        tempDir = File(System.getProperty("java.io.tmpdir"), UUID.randomUUID().toString())
        tempDir.mkdirs()
    }

    @After
    fun tearDown() {
        tempDir.deleteRecursively()
    }

    // Context Creation

    @Test
    fun testContextCreation() {
        val config = LexiconConfig(
            modelDir = File(tempDir, "models").absolutePath,
            dataDir = File(tempDir, "data").absolutePath
        )

        val context = LexiconContext.create(mockContext(), config)
        assertNotNull(context)
        context.close()
    }

    @Test(expected = LexiconException::class)
    fun testContextCreationWithInvalidPath() {
        val config = LexiconConfig(
            modelDir = "/nonexistent/path",
            dataDir = "/nonexistent/path"
        )

        LexiconContext.create(mockContext(), config)
    }

    // Error Mapping

    @Test
    fun testErrorMapping() {
        // Verify Rust errors map to Kotlin exceptions
        val config = LexiconConfig(
            modelDir = tempDir.absolutePath,
            dataDir = tempDir.absolutePath
        )
        val context = LexiconContext.create(mockContext(), config)

        try {
            runBlocking {
                context.loadModel("nonexistent")
            }
            fail("Should throw for nonexistent model")
        } catch (e: LexiconException) {
            // Expected
            assertTrue(e.message?.contains("model") == true)
        } finally {
            context.close()
        }
    }

    // Memory Management

    @Test
    fun testAutoCloseable() {
        val config = LexiconConfig(
            modelDir = tempDir.absolutePath,
            dataDir = tempDir.absolutePath
        )

        LexiconContext.create(mockContext(), config).use { context ->
            assertNotNull(context.availableModels())
        }
        // Context should be closed
    }

    @Test
    fun testRepeatedOperations() {
        val config = LexiconConfig(
            modelDir = tempDir.absolutePath,
            dataDir = tempDir.absolutePath
        )

        repeat(100) {
            LexiconContext.create(mockContext(), config).use { context ->
                context.availableModels()
            }
        }
        // Should not crash or leak
    }

    // Thread Safety

    @Test
    fun testConcurrentAccess() = runBlocking {
        val config = LexiconConfig(
            modelDir = tempDir.absolutePath,
            dataDir = tempDir.absolutePath
        )
        val context = LexiconContext.create(mockContext(), config)

        try {
            val jobs = (1..10).map {
                async {
                    context.availableModels()
                }
            }
            jobs.awaitAll()
        } finally {
            context.close()
        }
    }

    // Type Conversions

    @Test
    fun testSegmentConversion() {
        // Test that Segment data class correctly represents Rust struct
        val segment = Segment(
            startMs = 0,
            endMs = 1000,
            text = "Hello",
            confidence = 0.95f,
            speakerId = 1
        )

        assertEquals(0L, segment.startMs)
        assertEquals(1000L, segment.endMs)
        assertEquals("Hello", segment.text)
        assertEquals(0.95f, segment.confidence)
        assertEquals(1, segment.speakerId)
    }

    // Helper

    private fun mockContext(): android.content.Context {
        // Return mock Android context for tests
        // In real tests, use Robolectric or instrumented tests
        throw UnsupportedOperationException("Use Robolectric or instrumented tests")
    }
}
```

## Implementation Notes

### Mock Requirements

**Swift**: Use protocol-based mocking or dependency injection for platform services.

**Kotlin**: Use Robolectric for Android context mocking in unit tests. For integration tests, use instrumented tests on device/emulator.

### Error Handling

UniFFI maps Rust `Result<T, E>` to:
- **Swift**: `throws` functions with error types
- **Kotlin**: Functions that throw exceptions

Verify each error variant maps correctly:
- `LexiconError::Audio` → `LexiconError.audio`
- `LexiconError::Model` → `LexiconError.model`
- `LexiconError::Inference` → `LexiconError.inference`
- `LexiconError::Io` → `LexiconError.io`

### Memory Testing

**Swift**: Use `autoreleasepool` and weak references to verify deallocation.

**Kotlin**: Use JUnit `@After` cleanup and verify no references held.
